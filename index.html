  <script>

    const CALENDAR_URL = "https://rwcmd.asimut.net/api/ical.php?token=FposT5KcfVwdi39w";


    function parseICS(data) {

      const events = [];

      const regex = /BEGIN:VEVENT([\s\S]*?)END:VEVENT/g;

      let match;

      while ((match = regex.exec(data)) !== null) {

        const section = match[1];

        const getField = (k) => {

          const r = new RegExp(`${k}:(.*)`);

          const m = section.match(r);

          return m ? m[1].trim().replace(/\\n/g, '<br>') : "";

        };

        events.push({

          summary: getField("SUMMARY"),

          description: getField("DESCRIPTION"),

          start: getField("DTSTART"),

          end: getField("DTEND"),

          location: getField("LOCATION"),

          organiser: getField("ORGANIZER"),

          attendees: getField("ATTENDEE")

        });

      }

      return events;

    }


    function icsDateToObj(dateString) {

      if (!dateString) return null;

      try {

        if (!dateString.includes("T")) {

          const y = dateString.slice(0, 4), m = dateString.slice(4, 6), d = dateString.slice(6, 8);

          return new Date(`${y}-${m}-${d}T00:00:00`);

        } else {

          let ds = dateString.replace(/Z$/, "");

          let year = ds.substring(0, 4), month = ds.substring(4, 6), day = ds.substring(6, 8),

            hour = ds.substring(9, 11), min = ds.substring(11, 13);

          return new Date(Date.UTC(year, month-1, day, hour, min));

        }

      } catch { return null; }

    }


    function icsDateToTime(start, end) {

      const opts = { hour: "2-digit", minute: "2-digit" };

      const s = icsDateToObj(start), e = icsDateToObj(end);

      if (!s) return "";

      if (end && e) return `${s.toLocaleTimeString([], opts)} - ${e.toLocaleTimeString([], opts)}`;

      return s.toLocaleTimeString([], opts);

    }


    function formatDayHeader(dateObj) {

      if (!dateObj) return "";

      const options = { weekday: "long", day: "numeric", month: "long", year: "numeric" };

      return dateObj.toLocaleDateString(undefined, options);

    }


    async function loadICal() {

      const calDiv = document.getElementById("calendar");

      calDiv.innerHTML = "Loading...";

      try {

        let proxyUrl = "https://corsproxy.io/?" + encodeURIComponent(CALENDAR_URL);

        let res = await fetch(proxyUrl);

        if (!res.ok) throw Error("Network error.");

        let data = await res.text();

        let events = parseICS(data);

        const now = new Date();

        events = events

          .filter(ev => {

            const d = icsDateToObj(ev.start);

            return d && d >= now;

          })

          .sort((a, b) => (a.start > b.start ? 1 : -1));

        const grouped = events.reduce((acc, ev) => {

          const d = icsDateToObj(ev.start);

          if (!d) return acc;

          const ymd = d.toISOString().slice(0, 10);

          if (!acc[ymd]) acc[ymd] = [];

          acc[ymd].push(ev);

          return acc;

        }, {});

        let html = "";

        for (const day in grouped) {

          const dateObj = new Date(day + "T00:00:00");

          html += `<div class="agenda-day"><div class="agenda-day-header">${formatDayHeader(dateObj)}</div>`;

          grouped[day].forEach(ev => {

            html += `<div class="event-card">

                <div class="event-time">${icsDateToTime(ev.start, ev.end)}</div>

                <div class="event-title">${ev.summary || "Untitled Event"}</div>

                ${ev.location ? `<div class="event-detail"><strong>Location:</strong> ${ev.location}</div>` : ""}

                ${ev.description ? `<div class="event-detail">${ev.description}</div>` : ""}

              </div>`;

          });

          html += `</div>`;

        }

        calDiv.innerHTML = html || "<p>No upcoming events found in the feed.</p>";

      } catch {

        calDiv.innerHTML = `<span style="color:#e66465;">Failed to load calendar. Check the URL and availability.</span>`;

      }

    }


    window.onload = () => loadICal();

  </script>
